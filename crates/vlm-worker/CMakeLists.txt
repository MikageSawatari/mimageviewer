cmake_minimum_required(VERSION 3.24)
project(mimageviewer-vlm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This prototype requires Windows x64 and MSVC")
endif()
if(NOT DEFINED ENV{VULKAN_SDK})
    set(ENV{VULKAN_SDK} "C:/VulkanSDK/1.4.363.0")
endif()
option(VLM_REFERENCE_TOOLS "Build offline development reference tool (uses common)" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP_FETCH OFF CACHE BOOL "" FORCE)
foreach(flag SSE42 BMI2 AVX AVX2 AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AVX_VNNI FMA F16C)
    set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_BORINGSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_LIBRESSL OFF CACHE BOOL "" FORCE)
foreach(flag OPENSSL BUILD_SERVER BUILD_TOOLS BUILD_EXAMPLES BUILD_TESTS BUILD_APP SUBPROCESS)
    set(LLAMA_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_COMMON ${VLM_REFERENCE_TOOLS} CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
# Do not let the enclosing mIV checkout's Git identity label the vendored code.
set(LLAMA_BUILD_NUMBER 12032)
set(LLAMA_BUILD_COMMIT "dde879bc4f954b809892fd0d660515f42536394a")
get_filename_component(LLAMA_ROOT "../../vendor/llama.cpp" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory("${LLAMA_ROOT}" llama)
find_package(Vulkan REQUIRED)
file(GLOB VLM_IDENTITY_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(APPEND VLM_IDENTITY_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/source-identity.ps1")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${VLM_IDENTITY_FILES})
execute_process(COMMAND powershell -NoProfile -NonInteractive -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/source-identity.ps1" -SourceRoot "${CMAKE_CURRENT_SOURCE_DIR}"
    RESULT_VARIABLE VLM_IDENTITY_RESULT OUTPUT_VARIABLE VLM_SOURCE_SHA256
    ERROR_VARIABLE VLM_IDENTITY_ERROR OUTPUT_STRIP_TRAILING_WHITESPACE)
string(LENGTH "${VLM_SOURCE_SHA256}" VLM_IDENTITY_LENGTH)
if(NOT VLM_IDENTITY_RESULT EQUAL 0 OR NOT VLM_IDENTITY_LENGTH EQUAL 64 OR NOT VLM_SOURCE_SHA256 MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Cannot compute worker source identity: ${VLM_IDENTITY_ERROR}")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/build_identity.h"
    CONTENT "#pragma once\n#define MIV_VLM_SOURCE_SHA256 \"@VLM_SOURCE_SHA256@\"\n#define MIV_VLM_SOURCE_MARKER \"MIV_VLM_SOURCE_SHA256:@VLM_SOURCE_SHA256@\"\n" @ONLY)
add_executable(mimageviewer-vlm src/main.cpp src/devices.cpp src/inference.cpp src/serve.cpp)
target_include_directories(mimageviewer-vlm PRIVATE "${LLAMA_ROOT}/vendor" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(mimageviewer-vlm PRIVATE ggml llama mtmd Vulkan::Vulkan dxgi)
target_compile_definitions(mimageviewer-vlm PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(mimageviewer-vlm PRIVATE /utf-8 /W4 /permissive-)
get_filename_component(OUTPUT_DIR "../../vendor/vlm-worker" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(mimageviewer-vlm PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${OUTPUT_DIR}"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${OUTPUT_DIR}")
if(VLM_REFERENCE_TOOLS)
    add_executable(vlm-reference tools/reference.cpp)
    target_link_libraries(vlm-reference PRIVATE llama-common mtmd)
    target_include_directories(vlm-reference PRIVATE src)
    target_compile_options(vlm-reference PRIVATE /utf-8)
endif()
enable_testing()
add_executable(vlm-contract-tests tests/contracts.cpp)
target_include_directories(vlm-contract-tests PRIVATE src "${LLAMA_ROOT}/vendor")
target_compile_options(vlm-contract-tests PRIVATE /utf-8)
add_test(NAME vlm-contracts COMMAND vlm-contract-tests)
add_executable(vlm-greedy-tests tests/greedy.cpp)
target_include_directories(vlm-greedy-tests PRIVATE src)
target_link_libraries(vlm-greedy-tests PRIVATE llama)
target_compile_options(vlm-greedy-tests PRIVATE /utf-8)
add_test(NAME vlm-greedy-equivalence COMMAND vlm-greedy-tests)
add_executable(vlm-image-attempt-tests tests/image_attempt.cpp)
target_include_directories(vlm-image-attempt-tests PRIVATE src "${LLAMA_ROOT}/vendor" "${LLAMA_ROOT}/ggml/include")
target_compile_options(vlm-image-attempt-tests PRIVATE /utf-8)
add_test(NAME vlm-image-failure-boundary COMMAND vlm-image-attempt-tests)
add_executable(vlm-output-tests tests/exclusive_output.cpp)
target_include_directories(vlm-output-tests PRIVATE src "${LLAMA_ROOT}/vendor")
target_compile_definitions(vlm-output-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(vlm-output-tests PRIVATE /utf-8)
add_test(NAME vlm-exclusive-output-and-png COMMAND vlm-output-tests "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(vlm-image-input-tests tests/image_input.cpp)
target_include_directories(vlm-image-input-tests PRIVATE src "${LLAMA_ROOT}/vendor")
target_compile_options(vlm-image-input-tests PRIVATE /utf-8)
add_test(NAME vlm-bounded-png-input COMMAND vlm-image-input-tests)
add_executable(vlm-protocol-tests tests/protocol.cpp)
target_include_directories(vlm-protocol-tests PRIVATE src "${LLAMA_ROOT}/vendor")
target_compile_options(vlm-protocol-tests PRIVATE /utf-8)
add_test(NAME vlm-protocol COMMAND vlm-protocol-tests)
